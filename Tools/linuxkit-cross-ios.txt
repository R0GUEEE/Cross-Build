# Meson cross-file for building the ios-linuxkit engine for iOS/arm64.
#
# Derived from the recipe ios-linuxkit uses itself (app/xcode-meson.sh), with one
# deliberate difference: inside an Xcode build SDKROOT is already the iPhoneOS SDK,
# so a bare `-arch arm64` targets iOS. In a plain CI shell SDKROOT is unset and the
# same flags would silently target macOS, so the sysroot is passed explicitly.
# @IOS_SDK@ is replaced by the workflow with `xcrun --sdk iphoneos --show-sdk-path`.
#
# jit stays off: the native/AOT backend needs an entitlement a sideload lacks, and
# the threaded-code interpreter is the whole reason this works on a stock device.
#
# The two -D flags below are not optional, and both were arrived at from a real
# compile failure rather than guessed:
#
#   _XOPEN_SOURCE      the iPhoneOS SDK's ucontext.h refuses to declare the
#                      deprecated ucontext routines without it, and
#                      platform/native_fault.c pulls ucontext in.
#   _DARWIN_C_SOURCE   defining _XOPEN_SOURCE alone stops Darwin's cdefs.h from
#                      auto-enabling the full C level, which hides DT_REG/DT_DIR,
#                      fstatat/unlinkat/linkat and makedev/major/minor. Passing
#                      both keeps those visible while satisfying ucontext.h.
#
# Deliberately NOT done here: force-including <dispatch/dispatch.h> for
# platform/darwin.c. Meson applies `c_args` to assembly compilations as well, and
# the asbestos gadgets are .S files -- a force-included C header fed to the
# assembler produces "unrecognized instruction mnemonic" across the gadget sources.
# The workflow prepends that include to platform/darwin.c instead, which keeps it
# scoped to the one translation unit that needs it.

[binaries]
c = 'clang'
ar = 'ar'

[host_machine]
system = 'darwin'
cpu_family = 'aarch64'
cpu = 'aarch64'
endian = 'little'

[built-in options]
c_args = ['-arch', 'arm64', '-isysroot', '@IOS_SDK@', '-miphoneos-version-min=16.0',
          '-D_XOPEN_SOURCE=600', '-D_DARWIN_C_SOURCE']

[properties]
needs_exe_wrapper = true
